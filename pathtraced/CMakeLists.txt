cmake_minimum_required(VERSION 3.18)
project(arx_compact_pathtracer LANGUAGES CXX)
option(ARX_ENABLE_OPENMP "Use OpenMP when supported by the selected compiler" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(ARX_ENABLE_OPENMP)
  find_package(OpenMP QUIET COMPONENTS CXX)
endif()
add_executable(pathtrace source/pathtrace.cpp)
add_executable(test_tracer source/test_tracer.cpp)
foreach(target pathtrace test_tracer)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -O3)
  endif()
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endforeach()
if(OpenMP_CXX_FOUND)
  message(STATUS "ARX renderer: OpenMP enabled")
else()
  message(STATUS "ARX renderer: serial fallback (OpenMP unavailable or disabled)")
endif()
enable_testing()
add_test(NAME tracer_numerics COMMAND test_tracer "${CMAKE_CURRENT_SOURCE_DIR}/assets/native.bin" "${CMAKE_CURRENT_SOURCE_DIR}/assets/frame_inputs/00036.bin")
set_tests_properties(tracer_numerics PROPERTIES TIMEOUT 120)
